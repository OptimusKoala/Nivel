# Nivel 1.14 — Lot D : Frigo et suggestions — Plan d'implémentation

> **Pour les agents :** SOUS-COMPÉTENCE REQUISE — utiliser `superpowers:subagent-driven-development` (recommandé) ou `superpowers:executing-plans` pour exécuter ce plan tâche par tâche.

**But :** proposer chaque jour trois repas légers de saison dans l'onglet Repas, classés selon ce qu'on a dans le frigo, et loggeables en un tap.

**Architecture :** **une recette est un aliment.** Elle entre dans `foods.json` comme plat ordinaire avec sa composition dans `compositions.json` ; un nouveau `recipes.json` ne porte que ce que le catalogue ne sait pas dire — les mois de saison et la préparation. Conséquence : les kcal sortent de `MealEstimator`, la logger c'est `catalog.line(for:)`, et le journal l'affiche déjà correctement. Aucun moteur de calcul nouveau. Le frigo est une liste d'ids sur `UserProfile`, et le classement un calcul pur sans aléatoire.

**Pile :** Swift 6, SwiftUI, SwiftData, XCTest.

**Spec :** `docs/superpowers/specs/2026-08-13-nivel-1.14-design.md` §6.

**Dépendance :** le **lot A** (les ingrédients dont les recettes sont faites). Aucune sur B et C.

`Recipe.swift` :

```swift
// Recette de saison (spec v1.14 §6.2). Ne porte QUE ce que le catalogue
// d'aliments ne sait pas dire : les mois où la recette a du sens, et sa
// préparation. Le nom, l'emoji, les créneaux, les ingrédients et les kcal vivent
// sur le FoodItem correspondant — une seule vérité par donnée.

import Foundation

public struct Recipe: Codable, Identifiable, Hashable, Sendable {
    /// Id du `FoodItem` marqué `isRecipe` qui porte le reste.
    public let itemID: String
    /// Mois (1-12) où la recette est de saison.
    public let months: [Int]
    /// Trois à quatre lignes de préparation, au ton de Nivelito : on encourage,
    /// on ne donne pas d'ordre.
    public let steps: [String]

    public var id: String { itemID }

    public func isInSeason(month: Int) -> Bool { months.contains(month) }
}

/// Catalogue chargé une fois. Repli vide si le JSON est corrompu : la bande
/// d'idées disparaît, l'app continue — même règle que les six autres catalogues.
public struct RecipeCatalog: Sendable {
    public let recipes: [Recipe]
    public let byItemID: [String: Recipe]

    public init(recipes: [Recipe]) {
        self.recipes = recipes
        self.byItemID = Dictionary(recipes.map { ($0.itemID, $0) }, uniquingKeysWith: { a, _ in a })
    }

    public static let empty = RecipeCatalog(recipes: [])

    public static func load() throws -> RecipeCatalog {
        RecipeCatalog(recipes: try Catalogs.recipes())
    }
}
```

Un petit type pour la règle de mutation, testable sans vue :

```swift
/// Mutations du frigo. Réassignation COMPLÈTE du tableau à chaque fois : jamais
/// de mutation en place, règle SwiftData que le code se répète depuis la v1
/// (`remindersEnabled`) — une mutation en place ne déclenche pas la sauvegarde.
enum Pantry {
    static func toggle(_ itemID: String, on profile: UserProfile) {
        if profile.pantryItemIDs.contains(itemID) {
            profile.pantryItemIDs = profile.pantryItemIDs.filter { $0 != itemID }
        } else {
            profile.pantryItemIDs = profile.pantryItemIDs + [itemID]
        }
    }
    static func add(_ itemID: String, on profile: UserProfile) {
        guard !profile.pantryItemIDs.contains(itemID) else { return }
        profile.pantryItemIDs = profile.pantryItemIDs + [itemID]
    }
    static func clear(on profile: UserProfile) { profile.pantryItemIDs = [] }
}
```

```swift
// NivelCore/Tests/NivelCoreTests/RecipeSuggesterTests.swift
// Classement des suggestions (spec §6.4) : saison, créneau, couverture du frigo,
// kcal, puis rotation par le jour. Aucun aléatoire — donc entièrement testable.
import XCTest
@testable import NivelCore

final class RecipeSuggesterTests: XCTestCase {
    private var catalog: FoodCatalog!
    private var recipes: RecipeCatalog!
    private var calendar: Calendar!

    override func setUpWithError() throws {
        catalog = try FoodCatalog.load()
        recipes = try RecipeCatalog.load()
        calendar = Calendar(identifier: .gregorian)
        calendar.timeZone = TimeZone(identifier: "Europe/Paris")!
    }

    private func august13() -> Date {
        calendar.date(from: DateComponents(year: 2026, month: 8, day: 13))!
    }

    func testOnlyInSeasonAndInSlot() {
        let picks = RecipeSuggester.suggestions(
            date: august13(), slot: .dinner, pantry: [],
            recipes: recipes, catalog: catalog, calendar: calendar, limit: 3)
        XCTAssertEqual(picks.count, 3)
        for pick in picks {
            XCTAssertTrue(pick.recipe.isInSeason(month: 8), pick.recipe.itemID)
            XCTAssertTrue(catalog.byID[pick.recipe.itemID]!.slots.contains(.dinner))
        }
    }

    /// Frigo vide : la bande marche quand même, classée aux kcal. C'est le cas le
    /// plus fréquent au début, pas un cas dégradé.
    func testEmptyPantrySortsByCalories() {
        let picks = RecipeSuggester.suggestions(
            date: august13(), slot: .dinner, pantry: [],
            recipes: recipes, catalog: catalog, calendar: calendar, limit: 3)
        let kcals = picks.map(\.kcal)
        XCTAssertEqual(kcals, kcals.sorted())
        XCTAssertTrue(picks.allSatisfy { $0.missingCount == $0.ingredientCount })
    }

    /// Une recette dont on a TOUS les ingrédients passe devant une recette moins
    /// calorique dont il manque la moitié.
    func testFullCoverageWinsOverCalories() {
        let target = recipes.recipes.first { recipe in
            recipe.isInSeason(month: 8)
                && catalog.byID[recipe.itemID]!.slots.contains(.dinner)
        }!
        let fullPantry = Set(catalog.line(for: catalog.byID[target.itemID]!)
                                .components.map(\.itemID))
        let picks = RecipeSuggester.suggestions(
            date: august13(), slot: .dinner, pantry: fullPantry,
            recipes: recipes, catalog: catalog, calendar: calendar, limit: 3)
        XCTAssertEqual(picks.first?.recipe.itemID, target.itemID)
        XCTAssertEqual(picks.first?.missingCount, 0)
    }

    /// Le même jour donne le même classement : la bande ne bouge pas entre deux
    /// ouvertures de l'onglet.
    func testStableWithinADay() {
        let first = RecipeSuggester.suggestions(date: august13(), slot: .dinner, pantry: [],
                                                recipes: recipes, catalog: catalog,
                                                calendar: calendar, limit: 3)
        let second = RecipeSuggester.suggestions(date: august13(), slot: .dinner, pantry: [],
                                                 recipes: recipes, catalog: catalog,
                                                 calendar: calendar, limit: 3)
        XCTAssertEqual(first.map(\.recipe.itemID), second.map(\.recipe.itemID))
    }

    /// Mais deux jours différents ne montrent pas forcément la même chose : la
    /// rotation évite qu'un mois entier propose trois fois les mêmes plats.
    func testRotatesAcrossDays() {
        let month = (1...28).map { day in
            RecipeSuggester.suggestions(
                date: calendar.date(from: DateComponents(year: 2026, month: 8, day: day))!,
                slot: .dinner, pantry: [], recipes: recipes, catalog: catalog,
                calendar: calendar, limit: 3).map(\.recipe.itemID)
        }
        XCTAssertGreaterThan(Set(month.flatMap { $0 }).count, 3,
```

```swift
// NivelCore/Sources/NivelCore/RecipeSuggester.swift
// Classement des suggestions de repas (spec v1.14 §6.4). Pur et SANS aléatoire :
// la bande doit être identique à chaque ouverture d'un même jour, et testable.

import Foundation

/// Une suggestion prête à afficher : la recette, ses kcal calculées et l'état du
/// frigo la concernant.
public struct RecipeSuggestion: Sendable, Equatable {
    public let recipe: Recipe
    public let kcal: Int
    public let ingredientCount: Int
    public let missingCount: Int

    public var hasEverything: Bool { missingCount == 0 }
}

public enum RecipeSuggester {
    public static func suggestions(
        date: Date,
        slot: MealSlot,
        pantry: Set<String>,
        recipes: RecipeCatalog,
        catalog: FoodCatalog,
        calendar: Calendar,
        limit: Int = 3
    ) -> [RecipeSuggestion] {
        let month = calendar.component(.month, from: date)

        // Saison ET créneau. Repli : on relâche le MOIS avant le créneau — une
        // idée hors saison vaut mieux qu'une bande vide au 1ᵉʳ février.
        let inSlot = recipes.recipes.filter {
            catalog.byID[$0.itemID]?.slots.contains(slot) ?? false
        }
        let pool = inSlot.filter { $0.isInSeason(month: month) }
        let candidates = pool.isEmpty ? inSlot : pool
        guard !candidates.isEmpty else { return [] }

        // Rotation par le jour de l'année : sans elle, un mois entier proposerait
        // trois fois les mêmes plats. Décalage circulaire du pool AVANT le tri —
        // il ne départage donc que les ex æquo, sans jamais passer devant la
        // couverture du frigo.
        let dayOfYear = calendar.ordinality(of: .day, in: .year, for: date) ?? 0
        let offset = candidates.isEmpty ? 0 : dayOfYear % candidates.count
        let rotated = Array(candidates[offset...] + candidates[..<offset])

        let scored = rotated.map { recipe -> RecipeSuggestion in
            let components = catalog.byID[recipe.itemID]
                .map { catalog.line(for: $0).components } ?? []
            let ids = components.map(\.itemID)
            let missing = ids.filter { !pantry.contains($0) }.count
            let kcal = MealEstimator.kcal(
                lines: catalog.byID[recipe.itemID].map { [catalog.line(for: $0)] } ?? [],
                kcalPer100g: catalog.kcalPer100g)
            return RecipeSuggestion(recipe: recipe, kcal: kcal,
                                    ingredientCount: ids.count, missingCount: missing)
        }

        // Couverture d'abord (ce qu'on a déjà sous la main), kcal ensuite.
        // `enumerated` conserve l'ordre tourné pour les parfaits ex æquo : un tri
        // stable, donc un classement reproductible.
        return scored.enumerated()
            .sorted { left, right in
                if left.element.missingCount != right.element.missingCount {
                    return left.element.missingCount < right.element.missingCount
                }
                if left.element.kcal != right.element.kcal {
                    return left.element.kcal < right.element.kcal
                }
                return left.offset < right.offset
            }
            .prefix(limit)
            .map(\.element)
    }
}
```
